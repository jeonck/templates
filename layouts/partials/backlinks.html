{{/* Backlinks: every page that links to the current page with a markdown link. Matches on the raw markdown, so links must be written as absolute site paths, e.g. [Test Plan](/docs/testing-qa/test-plan/). ponytail: O(pages^2) scan on every build — fine for a few hundred pages. If the site grows past that, build the map once in a .Store on the home page. */}} {{- $target := .RelPermalink -}} {{- $bare := strings.TrimSuffix "/" $target -}} {{- $backlinks := slice -}} {{- range site.Pages -}} {{- if ne .RelPermalink $target -}} {{- $raw := .RawContent -}} {{- if or (in $raw (printf "](%s)" $target)) (in $raw (printf "](%s)" $bare)) -}} {{- $backlinks = $backlinks | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- with sort $backlinks "Title" -}}
